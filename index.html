2. Code source de la fonction

// Coordonnées au format Longitude,Latitude
const departLng = 2.3522;  // Paris (Lon)
const departLat = 48.8566; // Paris (Lat)
const arriveeLng = 4.8357; // Lyon (Lon)
const arriveeLat = 45.7640; // Lyon (Lat)

const url = `https://router.project-osrm.org/route/v1/driving/${departLng},${departLat};${arriveeLng},${arriveeLat}?steps=true&overview=false&language=fr`;

fetch(url)
  .then(response => response.json())
  .then(data => {
      if (data.code === 'Ok') {
          console.log("Itinéraire trouvé !");
          const etapes = data.routes[0].legs[0].steps;
          etapes.forEach((step, index) => {
              console.log(`${index + 1}. ${step.maneuver.instruction || step.maneuver.type} (${step.distance}m)`);
          });
      } else {
          console.error("Erreur OSRM :", data);
      }
  })
  .catch(err => console.error("Erreur réseau :", err));
